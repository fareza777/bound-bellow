  | "hand"
  | "shell"
  | "star"
  | "halo";

function drawMotif(ctx: CanvasRenderingContext2D, motif: Motif, accent: string): void {
  ctx.fillStyle = accent;
  ctx.strokeStyle = "rgba(20,13,9,0.8)";
  ctx.lineWidth = 1.15;
  ctx.lineCap = "round";
  switch (motif) {
    case "crown":
      ctx.beginPath();
      ctx.moveTo(-11, 6);
      ctx.lineTo(-8, -7);
      ctx.lineTo(-3, 0);
      ctx.lineTo(0, -10);
      ctx.lineTo(3, 0);
      ctx.lineTo(8, -7);
      ctx.lineTo(11, 6);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
      ctx.fillStyle = "rgba(30,20,12,0.8)";
      ctx.beginPath();
      ctx.arc(-5, 2, 1.1, 0, TAU);
      ctx.arc(0, 2, 1.1, 0, TAU);
      ctx.arc(5, 2, 1.1, 0, TAU);

      break;
    case "book":
      ctx.fillStyle = accent;
      ctx.fillRect(-10, -9, 20, 18);
      ctx.strokeRect(-10, -9, 20, 18);
      ctx.fillStyle = "rgba(30,20,12,0.7)";
      ctx.fillRect(-10, -9, 3.5, 18);
      ctx.strokeStyle = "rgba(30,20,12,0.55)";
      ctx.lineWidth = 0.8;
      for (let i = -5; i <= 5; i += 4) {
        ctx.beginPath();
        ctx.moveTo(-3, i);
        ctx.lineTo(7, i);

      ctx.strokeStyle = "rgba(234,223,200,0.45)";
      ctx.beginPath();
      ctx.moveTo(-6, -12);
      ctx.quadraticCurveTo(2, -10, 8, -12);
      ctx.stroke();
      break;
    case "root":
      ctx.strokeStyle = accent;
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.moveTo(0, 12);
      ctx.bezierCurveTo(-4, 3, -11, 2, -12, -8);
      ctx.moveTo(0, 4);
      ctx.bezierCurveTo(4, 0, 12, -2, 11, -10);
      ctx.moveTo(0, 7);
      ctx.bezierCurveTo(-3, 0, 2, -4, -1, -10);
      ctx.stroke();
      ctx.fillStyle = "rgba(234,223,200,0.5)";
      ctx.beginPath();
      ctx.arc(-11, -9, 1.4, 0, TAU);
      ctx.arc(10, -11, 1.4, 0, TAU);
      ctx.fill();
      break;
    case "needle":
      ctx.strokeStyle = "#e5d8bb";
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.moveTo(-11, 11);
      ctx.lineTo(10, -10);
      ctx.stroke();
      ctx.strokeStyle = accent;
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(-11, 11);
      ctx.bezierCurveTo(-4, 13, 3, 7, 8, 0);
      ctx.stroke();
      ctx.fillStyle = "#e5d8bb";
      ctx.beginPath();
      ctx.moveTo(10, -10);
      ctx.lineTo(7, -11);
      ctx.lineTo(11.5, -6.5);
      ctx.closePath();
      ctx.fill();
      break;
    case "candle":
      ctx.fillStyle = "#c9b28b";
      ctx.fillRect(-3, -1, 6, 13);
      ctx.strokeStyle = "rgba(30,20,12,0.7)";
      ctx.strokeRect(-3, -1, 6, 13);
      ctx.fillStyle = "rgba(201,178,139,0.65)";
      ctx.beginPath();
      ctx.ellipse(1.5, 10, 2.5, 4, 0, 0, TAU);
      ctx.fill();
      ctx.fillStyle = accent;
      ctx.beginPath();
      ctx.moveTo(0, -2);
      ctx.quadraticCurveTo(6, -8, 1, -14);
      ctx.quadraticCurveTo(-5, -8, 0, -2);
      ctx.fill();
      ctx.fillStyle = "#f7e1a6";
      ctx.beginPath();
      ctx.ellipse(0.5, -8, 1.2, 2.8, 0.1, 0, TAU);
      ctx.fill();
      break;
    case "bell":
      ctx.beginPath();
      ctx.moveTo(-8, 7);
      ctx.bezierCurveTo(-6, 1, -6, -5, 0, -9);
      ctx.bezierCurveTo(6, -5, 6, 1, 8, 7);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
      ctx.fillStyle = accent;
      ctx.beginPath();
      ctx.arc(0, -10, 2, 0, TAU);
      ctx.fill();
      ctx.fillStyle = "#3d2d1c";
      ctx.beginPath();
      ctx.arc(0, 9, 2, 0, TAU);
      ctx.fill();
      break;
    case "tooth":
      ctx.beginPath();
      ctx.moveTo(-8, -7);
      ctx.quadraticCurveTo(0, -12, 8, -7);
      ctx.quadraticCurveTo(6, 2, 3, 4);
      ctx.lineTo(2, 11);
      ctx.lineTo(-1, 5);
      ctx.lineTo(-3, 11);
      ctx.lineTo(-5, 4);
      ctx.quadraticCurveTo(-7, 2, -8, -7);
      ctx.closePath();
      ctx.fillStyle = "#d7cab0";
      ctx.fill();
      ctx.stroke();
      ctx.strokeStyle = "rgba(80,60,40,0.5)";
      ctx.beginPath();
      ctx.moveTo(-3, -6);
      ctx.lineTo(-1, 1);
      ctx.moveTo(3, -5);
      ctx.lineTo(1, 2);
      ctx.stroke();
      break;
    case "nail":
      ctx.strokeStyle = accent;
      ctx.lineWidth = 2.4;
      ctx.beginPath();
      ctx.moveTo(-10, 10);
      ctx.lineTo(7, -8);
      ctx.stroke();
      ctx.fillStyle = "#d7cab0";
      ctx.beginPath();

      ctx.lineTo(4, -9);
      ctx.lineTo(8, -5);
      ctx.closePath();
      ctx.fill();

      ctx.lineWidth = 1;
      ctx.stroke();

      for (const side of [-1, 1]) {
        ctx.save();
        ctx.scale(side, 1);
        ctx.beginPath();
        ctx.ellipse(7, -3, 9, 5, -0.4, 0, TAU);
        ctx.fill();
        ctx.stroke();
        ctx.fillStyle = "rgba(30,20,12,0.55)";

        ctx.fill();
        ctx.restore();

      ctx.fillStyle = "#5a4c3a";
      ctx.beginPath();
      ctx.ellipse(0, 2, 2.4, 8, 0, 0, TAU);
      ctx.fill();
      break;
    case "bottle":
      ctx.beginPath();
      ctx.moveTo(-4, -12);
      ctx.lineTo(4, -12);
      ctx.lineTo(4, -4);
      ctx.quadraticCurveTo(9, 0, 8, 7);
      ctx.quadraticCurveTo(3, 12, -8, 7);
      ctx.quadraticCurveTo(-9, 0, -4, -4);
      ctx.closePath();
      ctx.fill();

      ctx.fillStyle = "rgba(30,20,12,0.5)";
      ctx.beginPath();
      ctx.ellipse(0, 6, 5, 3, 0, 0, TAU);
      ctx.fill();
      ctx.fillStyle = "#c9b28b";
      ctx.fillRect(-3.5, -14, 7, 3);
      break;
    case "knot":
      ctx.strokeStyle = accent;
      ctx.lineWidth = 1.8;
      ctx.beginPath();
      ctx.moveTo(-11, 8);
      ctx.bezierCurveTo(-8, -2, 1, -11, 8, -5);
      ctx.bezierCurveTo(14, 0, 2, 9, -3, 4);
      ctx.bezierCurveTo(-9, -1, -2, -9, 5, -8);
      ctx.stroke();
      ctx.beginPath();
      ctx.arc(7, 9, 3, 0, TAU);
      ctx.stroke();
      break;
    case "bone":
      ctx.strokeStyle = "#d7cab0";
      ctx.lineWidth = 3.2;
      ctx.beginPath();
      ctx.moveTo(-8, 4);
      ctx.lineTo(8, -4);
      ctx.stroke();
      ctx.fillStyle = "#d7cab0";
      for (const [x, y] of [
        [-9, 2],
        [-7, 6],
        [9, -6],
        [7, -2],
      ]) {
        ctx.beginPath();
        ctx.arc(x, y, 2.2, 0, TAU);
        ctx.fill();
      }
      break;
    case "coin":
      ctx.beginPath();
      ctx.arc(0, 0, 10, 0, TAU);
      ctx.fill();
      ctx.stroke();
      ctx.fillStyle = "rgba(30,20,12,0.8)";
      ctx.beginPath();
      ctx.arc(0, 0, 3.2, 0, TAU);
      ctx.fill();
      ctx.strokeStyle = "rgba(255,235,190,0.6)";
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.arc(0, 0, 7, -2.6, -0.6);
      ctx.stroke();
      break;
    case "flame":
      ctx.beginPath();
      ctx.moveTo(0, 12);
      ctx.bezierCurveTo(-10, 4, -5, -2, -2, -7);
      ctx.quadraticCurveTo(-1, -1, 3, -4);
      ctx.quadraticCurveTo(7, -9, 4, -14);
      ctx.quadraticCurveTo(14, -6, 7, 4);
      ctx.quadraticCurveTo(4, 10, 0, 12);
      ctx.fill();
      ctx.stroke();
      ctx.fillStyle = "#f7e1a6";
      ctx.beginPath();
      ctx.ellipse(1, 2, 2.2, 5, 0.2, 0, TAU);
      ctx.fill();
      break;
    case "mirror":
      ctx.beginPath();
      ctx.ellipse(0, -1, 8, 11, 0, 0, TAU);
      ctx.fill();
      ctx.stroke();
      ctx.strokeStyle = "rgba(255,235,210,0.55)";
      ctx.lineWidth = 1.2;
      ctx.beginPath();
      ctx.arc(0, -1, 5.5, -2.5, -0.7);
      ctx.stroke();
      ctx.fillStyle = accent;
      ctx.fillRect(-2, 9, 4, 5);
      break;
    case "mask":
      ctx.beginPath();
      ctx.ellipse(0, 0, 8, 11, 0, 0, TAU);
      ctx.fillStyle = "#d7cab0";
      ctx.fill();
      ctx.stroke();
      ctx.fillStyle = "rgba(20,13,9,0.85)";
      ctx.beginPath();
      ctx.ellipse(-3.5, -3, 2.2, 3, 0, 0, TAU);
      ctx.ellipse(3.5, -3, 2.2, 3, 0, 0, TAU);
      ctx.fill();
      ctx.strokeStyle = "rgba(20,13,9,0.7)";
      ctx.lineWidth = 0.9;
      ctx.beginPath();
      ctx.moveTo(-4, 5);
      ctx.lineTo(4, 5);
      ctx.stroke();
      for (let i = -3; i <= 3; i += 2) {
        ctx.beginPath();
        ctx.moveTo(i, 3.5);
        ctx.lineTo(i, 6.5);
        ctx.stroke();
      }
      break;
    case "eye":
      ctx.beginPath();
      ctx.ellipse(0, 0, 11, 6, 0, 0, TAU);
      ctx.fillStyle = "#d7cab0";
      ctx.fill();
      ctx.stroke();
      ctx.fillStyle = accent;
      ctx.beginPath();
      ctx.arc(0, 0, 4, 0, TAU);
      ctx.fill();
      ctx.fillStyle = "#120d0a";
      ctx.beginPath();
      ctx.arc(0, 0, 1.7, 0, TAU);
      ctx.fill();
      ctx.strokeStyle = "rgba(20,13,9,0.7)";
      ctx.beginPath();
      ctx.moveTo(-11, 0);
      ctx.lineTo(-7, 0);
      ctx.moveTo(7, 0);
      ctx.lineTo(11, 0);
      ctx.stroke();
      break;
    case "spool":
      ctx.fillStyle = "#6f4b32";
      ctx.fillRect(-7, -10, 14, 20);
      ctx.fillStyle = accent;
      for (let y = -8; y < 9; y += 3.5) {
        ctx.fillRect(-7, y, 14, 2);
      }
      ctx.strokeStyle = "rgba(20,13,9,0.8)";
      ctx.strokeRect(-7, -10, 14, 20);
      ctx.beginPath();
      ctx.moveTo(7, 5);
      ctx.bezierCurveTo(13, 8, 8, 13, 3, 10);
      ctx.stroke();
      break;
    case "drop":
      ctx.beginPath();
      ctx.moveTo(0, -12);
      ctx.bezierCurveTo(8, -2, 9, 4, 0, 10);
      ctx.bezierCurveTo(-9, 4, -8, -2, 0, -12);
      ctx.fill();
      ctx.stroke();
      ctx.fillStyle = "rgba(255,240,215,0.45)";
      ctx.beginPath();
      ctx.ellipse(-2, 1, 1.5, 3, 0.3, 0, TAU);
      ctx.fill();
      break;
    case "button":
      ctx.beginPath();
      ctx.arc(0, 0, 9, 0, TAU);
      ctx.fill();
      ctx.stroke();
      ctx.fillStyle = "rgba(20,13,9,0.8)";
      for (const [x, y] of [
        [-3, -3],
        [3, -3],
        [-3, 3],
        [3, 3],
      ]) {
        ctx.beginPath();
        ctx.arc(x, y, 1.1, 0, TAU);
        ctx.fill();
      }
      ctx.strokeStyle = "rgba(20,13,9,0.5)";
      ctx.lineWidth = 0.8;
      ctx.beginPath();
      ctx.moveTo(-3, -3);
      ctx.lineTo(3, 3);
      ctx.moveTo(3, -3);
      ctx.lineTo(-3, 3);
      ctx.stroke();
      break;
    case "dice":
      ctx.fillStyle = "#d7cab0";
      ctx.fillRect(-9, -9, 18, 18);
      ctx.strokeRect(-9, -9, 18, 18);
      ctx.fillStyle = "rgba(20,13,9,0.85)";
      for (const [x, y] of [
        [-4, -4],
        [4, -4],
        [0, 0],
        [-4, 4],
        [4, 4],
      ]) {
        ctx.beginPath();
        ctx.arc(x, y, 1.4, 0, TAU);
        ctx.fill();
      }
      break;
    case "hand":
      ctx.fillStyle = "#d7cab0";
      ctx.beginPath();
      ctx.ellipse(0, 4, 7, 8, 0, 0, TAU);
      ctx.fill();
      ctx.stroke();
      for (let i = -2; i <= 2; i++) {
        ctx.beginPath();
        ctx.ellipse(i * 3, -4, 1.7, 5.5, i * 0.08, 0, TAU);
        ctx.fill();
      }
      ctx.strokeStyle = accent;
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(-5, 4);
      ctx.lineTo(5, 4);
      ctx.stroke();
      break;
    case "shell":
      ctx.beginPath();
      ctx.arc(0, 2, 10, Math.PI, TAU);
      ctx.lineTo(10, 5);
      ctx.quadraticCurveTo(0, 11, -10, 5);
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
      ctx.strokeStyle = "rgba(20,13,9,0.6)";
      ctx.lineWidth = 0.8;
      for (let i = 1; i <= 3; i++) {
        ctx.beginPath();
        ctx.arc(0, 2, i * 2.6, Math.PI, TAU);
        ctx.stroke();
      }
      break;
    case "star":
      ctx.beginPath();
      for (let i = 0; i < 10; i++) {
        const a = -Math.PI / 2 + (i / 10) * TAU;
        const r = i % 2 === 0 ? 11 : 4.5;
        const x = Math.cos(a) * r;
        const y = Math.sin(a) * r;
        if (i === 0) ctx.moveTo(x, y);
        else ctx.lineTo(x, y);
      }
      ctx.closePath();
      ctx.fill();
      ctx.stroke();
      break;
    case "halo":
      ctx.strokeStyle = accent;
      ctx.lineWidth = 2;
      ctx.beginPath();
      ctx.ellipse(0, -4, 9, 4, 0, 0, TAU);
      ctx.stroke();
      ctx.strokeStyle = "rgba(216,166,91,0.55)";
      ctx.lineWidth = 1;
      for (let i = 0; i < 8; i++) {
        const a = (i / 8) * TAU;
        ctx.beginPath();
        ctx.moveTo(Math.cos(a) * 11, -4 + Math.sin(a) * 5);
        ctx.lineTo(Math.cos(a) * 14, -4 + Math.sin(a) * 7);
        ctx.stroke();
      }
      break;
  }
}

function motifIcon(
  bg: string,
  accent: string,
  motif: Motif,
  kind: ItemKind = "relic",
): (ctx: CanvasRenderingContext2D, size: number) => void {
  return (ctx, s) => {
    tile(ctx, s, bg);
    ctx.save();
    ctx.translate(s / 2, s / 2);
    ctx.scale(s / 32, s / 32);
    drawMotif(ctx, motif, accent);
    ctx.restore();
    if (kind === "active") {
      ctx.strokeStyle = "rgba(216,166,91,0.8)";
      ctx.lineWidth = 1.5;
      ctx.strokeRect(2.5, 2.5, s - 5, s - 5);
      ctx.strokeStyle = "rgba(234,223,200,0.2)";
      ctx.lineWidth = 0.8;
      ctx.strokeRect(4.5, 4.5, s - 9, s - 9);
    } else if (kind === "trinket") {
      ctx.strokeStyle = "rgba(126,166,160,0.75)";
      ctx.lineWidth = 1.4;
      ctx.strokeRect(2.5, 2.5, s - 5, s - 5);
    }
  };
}
